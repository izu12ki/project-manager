# Slack 個人通知（DM） Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** アプリ内通知の受信者のうち、個人設定で Slack 通知をオンにした人に Slack ボットから DM を送る。プロジェクト単位の Webhook 投稿は削除する。

**Architecture:** `notify` が `planNotifications` で決めた受信者ごとの通知（1 人 1 件）を `user_integration` で絞り、`notifications/channels/` の `UserChannelProvider`（今は Slack のみ）で送る。ユーザーと Slack アカウントはオンにした時点で `users.lookupByEmail` により結びつけ、Slack ユーザー ID を保存する。Bot トークンは Worker シークレット `SLACK_BOT_TOKEN`。

**Tech Stack:** SvelteKit (Svelte 5) / Cloudflare Workers / Drizzle ORM + libSQL (Turso) / valibot / Vitest / Playwright

**Spec:** `docs/superpowers/specs/2026-10-14-slack-dm-design.md`

## Global Constraints

- 作業ディレクトリは `apps/web`（docs は `apps/docs`）。コマンドは `apps/web` で実行する。
- 検証: `pnpm check`（0 errors）、`pnpm lint`、`pnpm test:unit -- --run`。E2E は Task 5 と Task 6 で `pnpm test:e2e`。
- コミットはタスクごと。メッセージ形式 `feat(web): ...` / `refactor(web): ...` / `docs: ...`。自分が触っていない変更はステージしない。
- 既存マイグレーション（`drizzle/20261014023030_notifications`）は書き換えない。マイグレーションは `pnpm db:generate --name <name>` で生成する。
- Slack API: `https://slack.com/api/`、`Authorization: Bearer <token>`、タイムアウト 5 秒、再送なし。
- 文言（そのまま使う）:
  - 説明文: `このアプリのログイン用メールアドレス（{email}）と同じメールアドレスの Slack アカウントに、DM で通知が届きます。Slack のプロフィールのメールアドレスが異なる場合は届きません。`
  - 未設定: `Slack 連携は未設定です（管理者に連絡してください）`
  - 見つからない: `Slack に {email} のアカウントが見つかりませんでした。Slack のプロフィールのメールアドレスを確認してください。`
  - 設定不備: `Slack 連携の設定に問題があります。管理者に連絡してください。`
  - オン成功トースト: `Slack 通知をオンにしました。Slack に確認のメッセージを送りました。`
  - オフ成功トースト: `Slack 通知をオフにしました`
  - 確認 DM: `プロジェクト管理の通知をここで受け取ります。`
  - DM 1 行目: `assigned` → `{name}さんが あなたを担当者に設定 しました。` / `mentioned` → `{name}さんが あなたを メンション しました。` / `commented` → `{name}さんが コメント しました。` / `status_changed` → `{name}さんが ステータスを変更 しました。`
- コード内コメントは日本語で、既存の密度に合わせる。

## Review Focus

1. オンにする操作でメールが Slack 側と大文字小文字だけ違う → Slack は大文字小文字を区別しないので、そのまま送ってよい（加工しない）。メールに `+` を含む → `encodeURIComponent` で壊れないこと（Task 2 のテスト）。
2. 同じ人が 2 回オンにする（ID を引き直す） → 主キー衝突で落ちず、`external_id` が更新される（Task 3 のテスト）。
3. Slack が HTTP 200 で `{"ok":false,"error":"channel_not_found"}` を返す → 送信失敗としてログに出す（成功扱いしない）（Task 2 のテスト）。
4. 1 回の操作でコメント＋メンション → 1 人 1 件（優先度の高い `mentioned`）だけ DM が届く（Task 4 のテスト）。
5. 課題本文やタイトルに `<!channel>` や `<https://evil|x>` → エスケープされ、全員通知やリンク偽装にならない（Task 2 のテスト）。

```ts
export interface ChannelConfig {
	fetch: typeof fetch;
	token: string;
}

export type LookupResult =
	| { ok: true; externalId: string }
	/** not_found: その人が見つからない / config: トークン・権限・通信の問題 */
	| { ok: false; reason: 'not_found' | 'config'; error: string };

/** プロバイダに渡す解決済みの 1 通。1 行目は headline() で作る */
export interface DirectMessage {
	type: NotificationType;
	actor: { name: string };
	/** key は "ACME-12" */
	issue: { key: string; title: string; url: string };
	/** 2 行目。空なら出さない */
	detail: string;
}

/** 個人宛ての外部通知先。新しいサービスはこれを実装して index.ts に 1 行足す */
export interface UserChannelProvider {
	id: IntegrationProviderId;
	label: string;
	/** メールアドレスから相手を特定する。例外は投げない */
	lookup(cfg: ChannelConfig, email: string): Promise<LookupResult>;
	/** 1 件の通知を送る。例外は投げない */
	send(cfg: ChannelConfig, externalId: string, m: DirectMessage): Promise<SendResult>;
	/** オンにした直後の確認メッセージ。例外は投げない */
```

```ts
import { PRIORITY_LABEL, STATUS_LABEL, type Priority } from '$lib/labels';
import type { IssueEvent } from '../events';
import { excerpt } from '../format';
import type { PlannedNotification } from '../recipients';
import type { DirectMessage } from './types';

export interface DmSource {
	baseUrl: string;
	projectKey: string;
	issue: { number: number; title: string; priority: Priority };
	actorName: string;
	events: IssueEvent[];
}

const VERB: Record<DirectMessage['type'], string> = {
	assigned: 'あなたを担当者に設定',
	mentioned: 'あなたを メンション',
	commented: 'コメント',
	status_changed: 'ステータスを変更'
};

export function headline(m: DirectMessage): string {
	return `${m.actor.name}さんが ${VERB[m.type]} しました。`;
}

/** commentId（null は説明）→ その本文 */
function texts(events: IssueEvent[]): Map<string | null, string> {
	const map = new Map<string | null, string>();
	for (const e of events) {
		if (e.kind === 'commented') map.set(e.commentId, e.body);
		else if (e.kind === 'mentions_added') map.set(e.commentId, e.after);
		else if (e.kind === 'issue_created') map.set(null, e.description);
	}
	return map;
}

export function toDirectMessage(src: DmSource, n: PlannedNotification): DirectMessage {
	const base = `${src.baseUrl}/projects/${src.projectKey}/issues/${src.issue.number}`;
	let detail = '';
	switch (n.type) {
		case 'assigned':
			detail = `優先度: ${PRIORITY_LABEL[src.issue.priority]}`;
			break;
		case 'status_changed':
			if (n.fromStatus && n.toStatus)
				detail = `${STATUS_LABEL[n.fromStatus]} → ${STATUS_LABEL[n.toStatus]}`;
			break;
		case 'commented':
		case 'mentioned':
			detail = excerpt(texts(src.events).get(n.commentId) ?? '', 200);
			break;
	}
	return {
		type: n.type,
		actor: { name: src.actorName },
		issue: {
			key: `${src.projectKey}-${src.issue.number}`,
			title: src.issue.title,
			url: n.commentId ? `${base}#comment-${n.commentId}` : base
		},
		detail
	};
}
```

`slack.ts`:

```ts
import { headline } from './message';
import type { ChannelConfig, DirectMessage, SendResult, UserChannelProvider } from './types';

const API = 'https://slack.com/api/';

/** Slack の mrkdwn で制御文字になる & < > をエスケープする（<!channel> やリンク偽装を無効にする） */
const esc = (s: string) => s.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');

type SlackResponse = { ok: boolean; error?: string; user?: { id: string } };

/** Web API を呼ぶ。HTTP エラー・タイムアウト・例外は ok: false にする */
async function call(cfg: ChannelConfig, path: string, body?: unknown): Promise<SlackResponse> {
	try {
		const res = await cfg.fetch(API + path, {
			method: body === undefined ? 'GET' : 'POST',
			headers: {
				Authorization: `Bearer ${cfg.token}`,
				...(body === undefined ? {} : { 'Content-Type': 'application/json; charset=utf-8' })
			},
			body: body === undefined ? undefined : JSON.stringify(body),
			signal: AbortSignal.timeout(5000)
		});
		if (!res.ok) return { ok: false, error: `HTTP ${res.status}` };
		return (await res.json()) as SlackResponse;
	} catch (e) {
		if (e instanceof DOMException && e.name === 'TimeoutError') return { ok: false, error: 'タイムアウト' };
		return { ok: false, error: e instanceof Error ? e.message : String(e) };
	}
}

export function buildSlackDm(m: DirectMessage) {
	const head = headline(m);
	const ref = `${m.issue.key} ${m.issue.title}`;
	return {
		text: esc(`${head} ${ref}`),
		blocks: [
			{ type: 'section', text: { type: 'mrkdwn', text: `${esc(head)}\n*<${m.issue.url}|${esc(ref)}>*` } },
			...(m.detail
				? [{ type: 'context', elements: [{ type: 'mrkdwn', text: esc(m.detail) }] }]
				: [])
		]
	};
}

async function post(cfg: ChannelConfig, channel: string, payload: object): Promise<SendResult> {
	const r = await call(cfg, 'chat.postMessage', { channel, ...payload });
	return r.ok ? { ok: true } : { ok: false, error: r.error ?? 'unknown' };
}

export const slack: UserChannelProvider = {
	id: 'slack',
	label: 'Slack',
	async lookup(cfg, email) {
		const r = await call(cfg, `users.lookupByEmail?email=${encodeURIComponent(email)}`);
		if (r.ok && r.user) return { ok: true, externalId: r.user.id };
		const error = r.error ?? 'unknown';
		return { ok: false, reason: error === 'users_not_found' ? 'not_found' : 'config', error };
	},
	send: (cfg, externalId, m) => post(cfg, externalId, buildSlackDm(m)),
	sendWelcome: (cfg, externalId) =>
		post(cfg, externalId, { text: 'プロジェクト管理の通知をここで受け取ります。' })
};
```

```ts
import { and, eq } from 'drizzle-orm';
import type { AppContext } from '../context';
import { userIntegration } from '../db/schema';
import { AppError } from '../errors';
import { channelConfig, providers } from '../notifications/channels';

const mine = (ctx: AppContext) =>
	and(eq(userIntegration.userId, ctx.actor.id), eq(userIntegration.provider, 'slack'));

export async function getSlackStatus(
	ctx: AppContext
): Promise<{ available: boolean; linked: boolean }> {
	const [row] = await ctx.db.select({ id: userIntegration.externalId }).from(userIntegration).where(mine(ctx));
	return { available: channelConfig(ctx, 'slack') !== null, linked: !!row };
}

/** ログイン用メールアドレスで Slack のユーザーを探して保存し、確認 DM を送る */
export async function enableSlack(ctx: AppContext): Promise<void> {
	const cfg = channelConfig(ctx, 'slack');
	if (!cfg) throw new AppError('validation', 'Slack 連携は未設定です（管理者に連絡してください）');
	const email = ctx.actor.email;
	const found = await providers.slack.lookup(cfg, email);
	if (!found.ok) {
		if (found.reason === 'not_found')
			throw new AppError(
				'validation',
				`Slack に ${email} のアカウントが見つかりませんでした。Slack のプロフィールのメールアドレスを確認してください。`
			);
		console.error(`[slack] lookup failed: ${found.error}`);
		throw new AppError('validation', 'Slack 連携の設定に問題があります。管理者に連絡してください。');
	}
	await ctx.db
		.insert(userIntegration)
		.values({ userId: ctx.actor.id, provider: 'slack', externalId: found.externalId })
		.onConflictDoUpdate({
			target: [userIntegration.userId, userIntegration.provider],
			set: { externalId: found.externalId }
		});
	const r = await providers.slack.sendWelcome(cfg, found.externalId);
	if (!r.ok) console.error(`[slack] welcome failed ${ctx.actor.id}: ${r.error}`);
}

export async function disableSlack(ctx: AppContext): Promise<void> {
	await ctx.db.delete(userIntegration).where(mine(ctx));
}
```

```ts
/** オンにしている受信者に DM を送る。送信はレスポンス後に行う */
async function sendDirectMessages(
	ctx: AppContext,
	input: NotifyInput,
	priority: Priority,
	planned: PlannedNotification[]
): Promise<void> {
	const cfg = channelConfig(ctx, 'slack');
	if (!cfg || planned.length === 0) return;
	const links = await ctx.db
		.select({ userId: userIntegration.userId, externalId: userIntegration.externalId })
		.from(userIntegration)
		.where(
			and(
				eq(userIntegration.provider, 'slack'),
				inArray(
					userIntegration.userId,
					planned.map((n) => n.userId)
				)
			)
		);
	if (links.length === 0) return;
	const byUser = new Map(planned.map((n) => [n.userId, n]));
	const src: DmSource = {
		baseUrl: ctx.baseUrl,
		projectKey: input.project.key,
		issue: { number: input.issue.number, title: input.issue.title, priority },
		actorName: ctx.actor.name,
		events: input.events
	};
	for (const l of links) {
		const m = toDirectMessage(src, byUser.get(l.userId)!);
		ctx.waitUntil(
			providers.slack.send(cfg, l.externalId, m).then((r) => {
				if (!r.ok) console.error(`[dm] slack ${l.userId}: ${r.error}`);
			})
		);
	}
}
```
